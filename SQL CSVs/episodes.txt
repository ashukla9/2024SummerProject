Here is the schema of the database:

Table: episode
Columns:
  - episode_id (TEXT)
  - podcast_id (TEXT)
  - podcast_name (TEXT)
  - episode_title (TEXT)
  - episode_description (TEXT)
  - video_name (TEXT)
  - creation_time (TEXT)
Primary Keys: 
- episode_id
Foreign Keys:
- podcast_id references podcast(podcast_id)
- video_name references episodevideos(video_name)

Table: episodeengagement
Columns:
  - episode_id (TEXT)
  - podcast_id (TEXT)
  - podcast_name (TEXT)
  - user_id (TEXT)
  - video_name (TEXT)
  - reaction (TEXT)
  - comment (TEXT)
  - shares (INTEGER)
  - impressions (INTEGER)
  - engagement_timestamp (INTEGER)
Primary Keys:
- episode_id
Foreign Keys:
- podcast_id references podcast(podcast_id)
- user_id references user(user_id)
- video_name references episodevideos(video_name)

Table: episodevideos
Columns:
  - video_id (TEXT)
  - video_name (TEXT)
  - type (TEXT)
  - summary (TEXT)
  - duration (INTEGER)
Primary Keys:
- video_id
Foreign Keys:

Table: episodevideos_frames
Columns:
  - frame_id (TEXT)
  - video_id (TEXT)
  - video_name (TEXT)
  - timestamp (INTEGER)
Primary Keys:
- frame_id
Foreign Keys:
- video_id references episodevideos(video_id)
- video_name references episodevideos(video_name)

Table: episodevideos_objects
Columns:
  - object_id (TEXT)
  - video_id (TEXT)
  - video_name (TEXT)
  - type (TEXT)
  - category (TEXT)
  - tracking_id (TEXT)
  - timestamp (INTEGER)
Primary Keys:
- object_id
Foreign Keys:
- video_id references episodevideos(video_id)
- timestamp references episodevideos_frames(timestamp)
- video_name references episodevideos(video_name)

Table: episodevideos_shots
Columns:
  - shot_id (TEXT)
  - video_id (TEXT)
  - video_name (TEXT)
  - description (TEXT)
  - start (INTEGER)
  - finish (INTEGER)
Primary Keys:
- episodevideos_shots
Foreign Keys:
- video_id references episodevideos(video_id)
- video_name references episodevideos(video_name)

Table: episodevideos_transcripts
Columns:
  - transcript_id (TEXT)
  - video_id (TEXT)
  - video_name (TEXT)
  - text (TEXT)
  - start (INTEGER)
  - finish (INTEGER)
Primary Keys:

Foreign Keys:
- video_id references episodevideos(video_id)
- video_name references episodevideos(video_name)

Table: podcast
Columns:
  - podcast_id (TEXT)
  - podcast_name (TEXT)
  - description (TEXT)
  - host_name (TEXT)
Primary Keys:
- podcast_id
Foreign Keys:

Columns:
  - user_id (TEXT)

Primary Keys:
- user_id
Foreign Keys:
