Here is the schema of the database:

Table: city
Columns:
  - City_ID (INT)
  - Official_Name (TEXT)
  - Status (TEXT)
  - Area_km_2 (REAL)
  - Population (REAL)
  - Census_Ranking (TEXT)
Primary Keys:
  - City_ID
Foreign Keys:

Table: farm
Columns:
  - Farm_ID (INT)
  - Year (INT)
  - Total_Horses (REAL)
  - Working_Horses (REAL)
  - Total_Cattle (REAL)
  - Oxen (REAL)
  - Bulls (REAL)
  - Cows (REAL)
  - Pigs (REAL)
  - Sheep_and_Goats (REAL)
Primary Keys:
  - Farm_ID
Foreign Keys:

Table: farm_competition
Columns:
  - Competition_ID (INT)
  - Year (INT)
  - Theme (TEXT)
  - Host_city_ID (INT)
  - Hosts (TEXT)
Primary Keys:
  - Competition_ID
Foreign Keys:
  - Host_city_ID references city(City_ID)

Table: competition_record
Columns:
  - Competition_ID (INT)
  - Farm_ID (INT)
  - Rank (INT)
Primary Keys:
  - Competition_ID
  - Farm_ID
Foreign Keys:
  - Farm_ID references farm(Farm_ID)
  - Competition_ID references farm_competition(Competition_ID)
